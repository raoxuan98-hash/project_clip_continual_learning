57	def _parse_epoch_overrides(spec):
58	    if not spec:
59	        return {}
60	    result = {}
61	    for item in spec.split(","):
62	        item = item.strip()
63	        if not item:
64	            continue
65	        if "=" not in item:
66	            raise ValueError(
67	                f"Invalid --dataset_epoch_overrides item '{item}', expected dataset=epochs"
68	            )
69	        name, value = item.split("=", 1)
70	        name = name.strip()
71	        try:
72	            epochs = int(value)
73	        except ValueError as exc:
74	            raise ValueError(f"Invalid epoch count for {name}: {value}") from exc
75	        if epochs <= 0:
76	            raise ValueError(f"Epoch count must be positive for {name}, got {epochs}")
77	        result[name] = epochs
78	    return result

81	def _parse_int_list(spec):
82	    if not spec:
83	        return []
84	    result = []
85	    for item in str(spec).replace(" ", "").split(","):
86	        if not item:
87	            continue
88	        value = int(item)
89	        if value <= 0:
90	            raise ValueError(f"Expected positive integer in list, got {value}")
91	        result.append(value)
92	    return result

95	def _resolve_task_train_iterations(args, task_datasets, steps_per_epoch):
96	    if args.train_budget_mode == "uniform":
97	        return int(args.iterations), None
98	    if args.train_budget_mode != "lada_epochs":
99	        raise ValueError(f"Unsupported train_budget_mode: {args.train_budget_mode}")
100	
101	    epoch_map = dict(LADA_16SHOT_EPOCHS)
102	    epoch_map.update(_parse_epoch_overrides(args.dataset_epoch_overrides))
103	    missing = [d for d in task_datasets if d not in epoch_map]
104	    if missing:
105	        raise ValueError(
106	            "Missing LADA epoch schedule for datasets: "
107	            + ", ".join(missing)
108	            + ". Use --dataset_epoch_overrides dataset=epochs,..."
109	        )
110	    epochs = max(epoch_map[d] for d in task_datasets)
111	    return int(epochs * steps_per_epoch), epochs

120	def _build_text_classifier(args, model, processor, frozen_model, global_class_names,
121	                           cached_seen_text_features, device):
122	    if args.text_classifier_mode == "current":
123	        return get_zeroshot_classifier(model, processor, global_class_names, device)
124	    if args.text_classifier_mode != "lada_hybrid":
125	        raise ValueError(f"Unsupported text_classifier_mode: {args.text_classifier_mode}")
126	
127	    frozen_classifier = get_zeroshot_classifier(frozen_model, processor, global_class_names, device)
128	    if cached_seen_text_features is None or cached_seen_text_features.numel() == 0:
129	        return frozen_classifier
130	    seen_count = cached_seen_text_features.shape[0]
131	    hybrid = frozen_classifier.clone()
132	    hybrid[:, :seen_count] = cached_seen_text_features.to(device).t()
133	    hybrid = hybrid / hybrid.norm(dim=0, keepdim=True)
134	    return hybrid

137	def _fit_spherical_gmm_memory(features, labels, k):
138	    """Fit compact per-class spherical GMM memory from currently accessible features."""
139	    from sklearn.mixture import GaussianMixture
140	
141	    features_np = features.detach().cpu().float().numpy()
142	    labels_cpu = labels.detach().cpu().long()
143	    memory = {}
144	    for cid in sorted(labels_cpu.unique().tolist()):
145	        class_features = features_np[labels_cpu.numpy() == cid]
146	        actual_k = max(1, min(int(k), len(class_features)))
147	        gmm = GaussianMixture(
148	            n_components=actual_k,
149	            covariance_type="spherical",
150	            random_state=42 + int(cid),
151	            reg_covar=1e-6,
152	        )
153	        gmm.fit(class_features)
154	        memory[int(cid)] = {
155	            "means": torch.from_numpy(gmm.means_).float(),
156	            "covariances": torch.from_numpy(gmm.covariances_).float(),
157	            "weights": torch.from_numpy(gmm.weights_).float(),
158	        }
159	    return memory

162	def _sample_spherical_gmm_memory(gmm_memory, samples_per_class, mode, device):
163	    """Sample or repeat pseudo-features from compact spherical GMM memory."""
164	    if not gmm_memory:
165	        raise ValueError("GMM replay requested but no GMM memory is available")
166	    pseudo_features = []
167	    pseudo_labels = []
168	    for cid in sorted(gmm_memory):
169	        entry = gmm_memory[cid]
170	        means = entry["means"].to(device)
171	        covariances = entry["covariances"].to(device)
172	        weights = entry["weights"].to(device)
173	        n_components = means.shape[0]
174	
175	        total = int(samples_per_class)
176	        counts = torch.clamp((weights * total).round().long(), min=1)
177	        diff = total - int(counts.sum().item())
178	        if diff != 0:
179	            counts[int(weights.argmax().item())] += diff
180	        counts = torch.clamp(counts, min=1)
181	
182	        for comp_idx in range(n_components):
183	            count = int(counts[comp_idx].item())
184	            mean = means[comp_idx]
185	            if mode == "gmm_sample":
186	                std = torch.sqrt(torch.clamp(covariances[comp_idx], min=1e-8))
187	                samples = mean.unsqueeze(0) + torch.randn(
188	                    count, mean.shape[0], device=device
189	                ) * std
190	            elif mode == "gmm_mean":
191	                samples = mean.unsqueeze(0).repeat(count, 1)
192	            else:
193	                raise ValueError(f"Unsupported GMM replay mode: {mode}")
194	            samples = F.normalize(samples, dim=-1)
195	            pseudo_features.append(samples)
196	            pseudo_labels.append(
197	                torch.full((count,), int(cid), dtype=torch.long, device=device)
198	            )
199	
200	    return torch.cat(pseudo_features, dim=0), torch.cat(pseudo_labels, dim=0)

203	def _build_center_replay_features(stats_dict, center_means, samples_per_class, device):
204	    """Build compact center replay pseudo-features for classifier-only fine-tuning."""
205	    pseudo_features = []
206	    pseudo_labels = []
207	    for cid in sorted(stats_dict):
208	        if center_means is not None and cid in center_means:
209	            centers = center_means[cid].to(device)
210	        else:
211	            centers = stats_dict[cid].mean.unsqueeze(0).to(device)
212	
213	        n_centers = centers.shape[0]
214	        per_center = max(1, int(samples_per_class) // n_centers)
215	        remainder = int(samples_per_class) - per_center * n_centers
216	        for center_idx in range(n_centers):
217	            count = per_center + (1 if center_idx < remainder else 0)
218	            samples = centers[center_idx].unsqueeze(0).repeat(count, 1)
219	            samples = F.normalize(samples, dim=-1)
220	            pseudo_features.append(samples)
221	            pseudo_labels.append(
222	                torch.full((count,), int(cid), dtype=torch.long, device=device)
223	            )
224	
225	    return torch.cat(pseudo_features, dim=0), torch.cat(pseudo_labels, dim=0)

228	def _compute_dataset_balanced_global_cov(stats_dict, history_class_names):
229	    offset = 0
230	    per_dataset_covs = []
231	    for hcn in history_class_names:
232	        n_classes = len(hcn)
233	        ds_cov = sum(stats_dict[cid].cov for cid in range(offset, offset + n_classes)) / n_classes
234	        per_dataset_covs.append(ds_cov)
235	        offset += n_classes
236	    return sum(per_dataset_covs) / len(per_dataset_covs)

239	def _print_lada_metrics(name, tracker, task_names):
240	    """Print LADA-style Transfer/Average/Last metrics from a full KxK matrix."""
241	    print("\n" + "-" * 110)
242	    print(f"[{name} LADA Metrics]")
243	    header_str = " | ".join([f"{h[:8]:<8}" for h in task_names])
244	    print(f"Metric     | {header_str} | [Average]")
245	    print("-" * 110)
246	
247	    per_task = tracker.calculate_per_task_metrics()
248	    transfer = [per_task[t]["transfer"] for t in task_names]
249	    average = [per_task[t]["average"] for t in task_names]
250	    last = [per_task[t]["last"] for t in task_names]
251	    summary = tracker.get_summary()
252	
253	    print(
254	        "Transfer  | "
255	        + " | ".join(f"{x:8.1f}" if i > 0 else f"{'N/A':>8}" for i, x in enumerate(transfer))
256	        + f" | [{summary['transfer']:.1f}]"
257	    )
258	    print(
259	        "Average   | "
260	        + " | ".join(f"{x:8.1f}" for x in average)
261	        + f" | [{summary['average']:.1f}]"
262	    )
263	    print(
264	        "Last      | "
265	        + " | ".join(f"{x:8.1f}" for x in last)
266	        + f" | [{summary['last']:.1f}]"
267	    )
268	    print("-" * 110)

282	def _atomic_write_json(path, payload):
283	    path = Path(path)
284	    path.parent.mkdir(parents=True, exist_ok=True)
285	    tmp_path = path.with_suffix(path.suffix + ".tmp")
286	    with tmp_path.open("w", encoding="utf-8") as handle:
287	        json.dump(payload, handle, indent=2)
288	    os.replace(tmp_path, path)

291	def _atomic_save_step_artifact(
292	    args,
293	    step_index,
294	    task_name,
295	    model,
296	    cached_seen_text_features,
297	    lr_rgda_classifier,
298	    global_stats_dict,
299	    global_center_means,
300	    dataset_balanced_global_cov,
301	    task_names,
302	    global_class_names,
303	    dataset_label_offsets,
304	    dataset_class_counts,
305	    current_num_classes,
306	    gmm_memory=None,
307	    rgda_stats_by_m=None,
308	):
309	    """Save an immutable post-task snapshot and publish an eval-ready job."""
310	    if not args.async_eval_dir:
311	        raise ValueError("--async_eval_dir is required when saving step artifacts")
312	
313	    base_dir = Path(args.async_eval_dir)
314	    artifact_root = base_dir / "artifacts"
315	    queue_dir = base_dir / "queue"
316	    artifact_root.mkdir(parents=True, exist_ok=True)
317	    queue_dir.mkdir(parents=True, exist_ok=True)
318	
319	    step_name = f"step_{step_index + 1:02d}_{task_name}"
320	    tmp_dir = artifact_root / f".{step_name}.tmp"
321	    final_dir = artifact_root / step_name
322	    if tmp_dir.exists():
323	        shutil.rmtree(tmp_dir)
324	    if final_dir.exists():
325	        shutil.rmtree(final_dir)
326	    tmp_dir.mkdir(parents=True)
327	
328	    artifact_path = tmp_dir / "artifact.pt"
329	    artifact = {
330	        "schema_version": 1,
331	        "created_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
332	        "step_index": step_index,
333	        "task_name": task_name,
334	        "task_names": task_names,
335	        "global_class_names": global_class_names,
336	        "dataset_label_offsets": dataset_label_offsets,
337	        "dataset_class_counts": dataset_class_counts,
338	        "current_num_classes": current_num_classes,
339	        "gmm_memory": gmm_memory,
340	        "args": _serializable_args(args),
341	        "model_state_dict": _cpu_state_dict(model.state_dict()),
342	        "cached_seen_text_features": (
343	            cached_seen_text_features.detach().cpu()
344	            if cached_seen_text_features is not None
345	            else None
346	        ),
347	        "global_stats_dict": global_stats_dict,
348	        "global_center_means": global_center_means,
349	        "dataset_balanced_global_cov": dataset_balanced_global_cov.detach().cpu(),
350	        "rgda_stats_by_m": rgda_stats_by_m,
351	        "lr_rgda": {
352	            "state_dict": _cpu_state_dict(lr_rgda_classifier.classifier.state_dict()),
353	            "num_classes": lr_rgda_classifier.num_classes,
354	            "class_ids": lr_rgda_classifier.class_ids,
355	            "config": {
356	                "rank": args.rgda_rank,
357	                "qda_reg_alpha1": args.rgda_alpha1,
358	                "qda_reg_alpha2": args.rgda_alpha2,
359	                "qda_reg_alpha3": args.rgda_alpha3,
360	                "temperature": 1.0,
361	                "M": args.num_centers,
362	            },
363	        },
364	    }
365	    torch.save(artifact, artifact_path)
366	
367	    os.replace(tmp_dir, final_dir)
368	    ready_payload = {
369	        "schema_version": 1,
370	        "step_index": step_index,
371	        "task_name": task_name,
372	        "artifact_path": str((final_dir / "artifact.pt").resolve()),
373	    }
374	    _atomic_write_json(queue_dir / f"{step_name}.ready.json", ready_payload)
375	    logging.info("Saved async eval artifact: %s", final_dir)

618	def main(args):
619	    if args.seed is not None:
620	        fix_random_seed(args.seed)
621	
622	    # ========== 1. 初始化 ==========
623	    logging.info("\n=== Initializing Incremental Learning ===")
624	    trainer = LoRANSPTrainer(args)
625	    model = trainer.model
626	    processor = trainer.processor
627	
628	    # 解析文本编码器调度的默认值
629	    if args.text_tuning_schedule is None:
630	        args.text_tuning_schedule = "always" if args.tune_text_encoder else "never"
631	
632	    # 验证文本调度参数
633	    if args.text_schedule_switch_task < 1:
634	        raise ValueError("--text_schedule_switch_task must be >= 1")
635	    if args.text_lr_scale_after_task < 0:
636	        raise ValueError("--text_lr_scale_after_task must be >= 0")
637	
638	    # 非 never 调度时强制启用文本 LoRA（否则调度无意义）
639	    if args.text_tuning_schedule == "never":
640	        args.tune_text_encoder = False
641	    else:
642	        args.tune_text_encoder = True
643	
644	    use_distillation = args.fd_weight > 0 or args.cd_weight > 0
645	    reference_loader = load_reference_dataset(args, trainer.model_pretrain,
646	                                              processor, args.device) if use_distillation else None
647	
648	    # 预收集所有数据集的类名，用于全局 ZS 分类器（text LoRA 每次 merge 后更新）
649	    global_class_names = []
650	    task_names = [d[0] for d in args.dataset_sequence]
651	    dataset_label_offsets = {}
652	    dataset_class_counts = {}
653	    next_label_offset = 0
654	    for task_datasets in args.dataset_sequence:
655	        for d_name in task_datasets:
656	            c_names = get_xtail_classnames(args.root, d_name, args.num_shots)
657	            dataset_label_offsets[d_name] = next_label_offset
658	            dataset_class_counts[d_name] = len(c_names)
659	            next_label_offset += len(c_names)
660	            global_class_names.extend(c_names)
661	
662	    # ========== 2. 增量学习循环 ==========
663	    history_class_names = []  # 记录所有已学类名列表的列表
664	    cached_seen_text_features = None
665	    artifact_num_centers = sorted(set([args.num_centers] + _parse_int_list(args.artifact_num_centers)))
666	    global_rgda_stats_by_m = {m: {} for m in artifact_num_centers}
667	    global_rgda_centers_by_m = {m: None for m in artifact_num_centers}
668	    dataset_balanced_cov_by_m = {}
669	    global_stats_dict = {}
670	    global_center_means = None  # 多中心 LR-RGDA 的类中心映射，用局部变量替代 main._global_center_means
671	    global_gmm_memory = {}
672	
673	    metrics_zs = ContinualLearningMetrics(task_names)
674	    metrics_rgda = ContinualLearningMetrics(task_names)
675	    metrics_ens = ContinualLearningMetrics(task_names)
676	    acc_matrix_lada = [] if args.enable_lada else None
677	    acc_matrix_lada_zs = [] if args.enable_lada else None
678	
679	    for i, task_datasets in enumerate(args.dataset_sequence):
680	        print(f"\n" + "=" * 50)
681	        print(f"=== Task {i+1}: {task_datasets} ===")
682	        print("=" * 50)
683	
684	        # --- 2a. 准备训练数据 ---
685	        train_loaders = []
686	        task_class_names = []
687	        for d_name in task_datasets:
688	            train_transform, test_transform = get_transforms(d_name)
689	            num_shots = None if args.full_shot else args.num_shots
690	            tr_loader, _, _, c_names = get_xtail_trainloader(
691	                root=args.root, dataset_name=d_name,
692	                transform_train=train_transform, transform_test=test_transform,
693	                num_shots=num_shots, batch_size=args.batch_size
694	            )
695	            train_loaders.append(tr_loader)
696	            task_class_names.extend(c_names)
697	
698	        merged_dataset = ConcatDataset([loader.dataset for loader in train_loaders])
699	        # 用于提取协方差的 loader 不打乱顺序
700	        cov_loader = DataLoader(merged_dataset, batch_size=args.batch_size, shuffle=False)
701	        merged_loader = DataLoader(merged_dataset, batch_size=args.batch_size, shuffle=True)
702	        task_train_iterations, task_epochs = _resolve_task_train_iterations(
703	            args, task_datasets, len(merged_loader))
704	        if task_epochs is None:
705	            logging.info(
706	                "Task %d training budget: %d iterations (uniform)",
707	                i + 1, task_train_iterations,
708	            )
709	        else:
710	            logging.info(
711	                "Task %d training budget: %d epochs x %d steps/epoch = %d iterations",
712	                i + 1, task_epochs, len(merged_loader), task_train_iterations,
713	            )
714	
715	        # --- 2b. 初始化适配器（非标准 LoRA 时在训练前初始化） ---
716	        if args.init_mode not in ["lora_nsp", "lora_vanilla"]:
717	            logging.info(f"\n=== Initializing adapters ({args.init_mode}) ===")
718	            if "proj_sigma" in args.init_mode and args.init_mode != "lora_nsp":
719	                window = "tail" if "tail" in args.init_mode else "middle"
720	                if trainer.covariance_history:
721	                    trainer.model.vision_model.initialize_adapters_from_covariance(
722	                        trainer.covariance_history, window=window)
723	                else:
724	                    logging.info("No covariance history yet, using default random init for Task 1")
725	            elif "weight_svd" in args.init_mode:
726	                window = "tail" if "tail" in args.init_mode else "middle"
727	                trainer.model.vision_model.initialize_adapters_from_weight_svd(window=window)
728	
729	        # --- 2c. 训练模型 ---
730	        # 文本编码器调度：根据当前任务决定 text_lr
731	        task_number = i + 1
732	        schedule = args.text_tuning_schedule
733	        text_lr = args.lr  # 默认
734	        if schedule == "never":
735	            text_lr = 0.0
736	        elif schedule in ("freeze_after", "low_lr_after"):
737	            if task_number > args.text_schedule_switch_task:
738	                if schedule == "freeze_after":
739	                    text_lr = 0.0
740	                else:
741	                    text_lr = args.lr * args.text_lr_scale_after_task
742	        train_text_this_task = text_lr > 0
743	        model = trainer.train(merged_loader, task_class_names, reference_loader,
744	                              aux_weight=args.aux_weight,
745	                              train_text_encoder=train_text_this_task,
746	                              text_lr=text_lr,
747	                              iterations=task_train_iterations)
748	
749	        # --- 2d. 任务后处理：合入 + 协方差累积 ---
750	        if args.init_mode == "lora_nsp":
751	            print("\n=== Applying Null-Space Projection (NSP) ===")
752	            # 图像编码器 NSP
753	            text_covariances = None
754	            if trainer.has_vision_lora:
755	                covariances = trainer.extract_layer_covariances(cov_loader)
756	            # 文本编码器 NSP（仅当本轮实际训练了文本编码器）
757	            if trainer.has_text_lora and train_text_this_task:
758	                text_covariances = trainer.extract_text_covariances(task_class_names)
759	            trainer.finalize_task_for_incremental()
760	            if trainer.has_vision_lora:
761	                trainer.update_covariance_history(covariances)
762	            if trainer.has_text_lora and train_text_this_task:
763	                trainer.update_text_covariance_history(text_covariances)
764	        elif "proj_sigma" in args.init_mode:
765	            print("\n=== Proj-Σ: Extracting covariances + merging ===")
766	            if trainer.has_vision_lora:
767	                covariances = trainer.extract_layer_covariances(cov_loader)
768	                trainer.update_covariance_history(covariances, update_projection=False)
769	            trainer.finalize_task_for_incremental()
770	        else:
771	            print(f"\n=== Merging LoRA Weights (init_mode={args.init_mode}) ===")
772	            trainer.finalize_task_for_incremental()
773	
774	        if args.text_classifier_mode == "lada_hybrid":
775	            task_text_features = _encode_text_classifier_columns(
776	                model, processor, task_class_names, args.device)
777	            cached_seen_text_features = (
778	                task_text_features.detach()
779	                if cached_seen_text_features is None
780	                else torch.cat([cached_seen_text_features, task_text_features.detach()], dim=0)
781	            )
782	            logging.info(
783	                "Cached LADA-style seen text prototypes: %s",
784	                tuple(cached_seen_text_features.shape),
785	            )
786	            if args.text_adapter_type == "lada_adaptformer" and hasattr(model.text_model, "reset_adapters"):
787	                model.text_model.reset_adapters()
788	                logging.info("Reset LADA text AdaptFormer after caching current task prototypes.")
789	
790	        # --- 2e. 提取特征并构建统计字典 ---
791	        task_features = []
792	        task_labels = []
793	        task_gmm_features = []
794	        task_gmm_labels = []
795	        label_offset = sum(len(c_names) for c_names in history_class_names)
796	
797	        for d_name in task_datasets:
798	            train_transform, test_transform = get_transforms(d_name)
799	            tr_loader, tr4update, _, c_names = get_xtail_trainloader(
800	                root=args.root, dataset_name=d_name,
801	                transform_train=train_transform, transform_test=test_transform,
802	                num_shots=num_shots, batch_size=args.batch_size
803	            )
804	            # 根据 --classifier_feature_transform 选择特征提取的 transform
805	            # 'test' = 确定性变换（publication protocol），'train' = 随机增强
806	            feature_loader = tr_loader if args.classifier_feature_transform == "train" else tr4update
807	            from src.utils.feature_extractor import extract_features
808	            raw_features, labels = extract_features(
809	                model, feature_loader, args.device, normalize=False)
810	            features = F.normalize(raw_features, dim=-1)
811	            global_labels = labels + label_offset
812	            task_features.append(features)
813	            task_labels.append(global_labels)
814	            gmm_features = raw_features if args.gmm_fit_space == "raw" else features
815	            task_gmm_features.append(gmm_features)
816	            task_gmm_labels.append(global_labels)
817	
818	        task_features = torch.cat(task_features)
819	        task_labels = torch.cat(task_labels)
820	        task_gmm_features = torch.cat(task_gmm_features)
821	        task_gmm_labels = torch.cat(task_gmm_labels)
822	
823	        # 构建一套或多套 compact LR-RGDA stats。主 M 用于 inline classifier；
824	        # 其它 M 只写入 artifact，供离线 classifier sweep 使用。
825	        for m in artifact_num_centers:
826	            task_stats_dict_m, center_means_m = build_multi_center_stats_dict(
827	                task_features, task_labels, M=m)
828	            global_rgda_stats_by_m[m].update(task_stats_dict_m)
829	            if center_means_m is not None:
830	                if global_rgda_centers_by_m[m] is None:
831	                    global_rgda_centers_by_m[m] = {}
832	                # center_means 的 key 已是全局类ID（来自 task_labels + label_offset）
833	                global_rgda_centers_by_m[m].update(center_means_m)
834	
835	        global_stats_dict = global_rgda_stats_by_m[args.num_centers]
836	        global_center_means = global_rgda_centers_by_m[args.num_centers]
837	
838	        task_gmm_memory = _fit_spherical_gmm_memory(
839	            task_gmm_features, task_gmm_labels, k=args.gmm_k)
840	        global_gmm_memory.update(task_gmm_memory)
841	        logging.info(
842	            "[GMM Memory] Updated spherical GMM memory: %d seen classes, "
843	            "k=%d, fit_space=%s",
844	            len(global_gmm_memory), args.gmm_k, args.gmm_fit_space,
845	        )
846	
847	        history_class_names.append(task_class_names)
848	
849	        # --- 2f. 构建分类器 ---
850	        # 计算数据集等权的全局协方差
851	        dataset_balanced_cov_by_m = {
852	            m: _compute_dataset_balanced_global_cov(stats_dict_m, history_class_names)
853	            for m, stats_dict_m in global_rgda_stats_by_m.items()
854	        }
855	        dataset_balanced_global_cov = dataset_balanced_cov_by_m[args.num_centers]
856	
857	        lr_rgda_classifier = LRRGDAClassifier(
858	            stats_dict=global_stats_dict,
859	            device=args.device,
860	            rank=args.rgda_rank,
861	            M=args.num_centers,
862	            center_means=global_center_means,
863	            qda_reg_alpha1=args.rgda_alpha1,
864	            qda_reg_alpha2=args.rgda_alpha2,
865	            qda_reg_alpha3=args.rgda_alpha3,
866	            temperature=1.0,
867	            global_cov=dataset_balanced_global_cov,
868	        )
869	        # 分类器微调（如果启用）：LR-RGDA 仍由正常 mean/cov 构建；
870	        # 这里只使用 compact replay pseudo-features 微调 classifier 参数。
871	        if args.rgda_train_iter > 0:
872	            if args.rgda_fit_source == "none":
873	                logging.info(
874	                    "[Fit] --rgda_train_iter=%d but --rgda_fit_source=none; "
875	                    "skipping classifier fine-tuning.",
876	                    args.rgda_train_iter,
877	                )
878	                replay_features = replay_labels = None
879	            elif args.rgda_fit_source == "center_replay":
880	                replay_features, replay_labels = _build_center_replay_features(
881	                    global_stats_dict,
882	                    global_center_means,
883	                    args.gaussian_samples_per_class,
884	                    args.device,
885	                )
886	            elif args.rgda_fit_source in ("gmm_mean", "gmm_sample"):
887	                replay_features, replay_labels = _sample_spherical_gmm_memory(
888	                    global_gmm_memory,
889	                    args.gaussian_samples_per_class,
890	                    args.rgda_fit_source,
891	                    args.device,
892	                )
893	            else:
894	                raise ValueError(f"Unsupported rgda_fit_source: {args.rgda_fit_source}")
895	
896	            if replay_features is not None:
897	                logging.info(
898	                    "[Fit] Fine-tuning LR-RGDA for %d iters on %d compact replay "
899	                    "pseudo-features (fit_source=%s, samples_per_class=%d)...",
900	                    args.rgda_train_iter,
901	                    replay_features.shape[0],
902	                    args.rgda_fit_source,
903	                    args.gaussian_samples_per_class,
904	                )
905	                lr_rgda_classifier.fit(
906	                    replay_features, replay_labels,
907	                    iterations=args.rgda_train_iter,
908	                    lr=args.rgda_train_lr,
909	                )
910	
911	        current_num_classes = sum(len(c_names) for c_names in history_class_names)
912	        if args.save_step_artifacts:
913	            rgda_stats_by_m = {
914	                int(m): {
915	                    "global_stats_dict": global_rgda_stats_by_m[m],
916	                    "global_center_means": global_rgda_centers_by_m[m],
917	                    "dataset_balanced_global_cov": dataset_balanced_cov_by_m[m].detach().cpu(),
918	                }
919	                for m in artifact_num_centers
920	            }
921	            _atomic_save_step_artifact(
922	                args=args,
923	                step_index=i,
924	                task_name=task_datasets[0],
925	                model=model,
926	                cached_seen_text_features=cached_seen_text_features,
927	                lr_rgda_classifier=lr_rgda_classifier,
