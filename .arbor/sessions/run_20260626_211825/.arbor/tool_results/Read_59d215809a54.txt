57	def _parse_epoch_overrides(spec):
58	    if not spec:
59	        return {}
60	    result = {}
61	    for item in spec.split(","):
62	        item = item.strip()
63	        if not item:
64	            continue
65	        if "=" not in item:
66	            raise ValueError(
67	                f"Invalid --dataset_epoch_overrides item '{item}', expected dataset=epochs"
68	            )
69	        name, value = item.split("=", 1)
70	        name = name.strip()
71	        try:
72	            epochs = int(value)
73	        except ValueError as exc:
74	            raise ValueError(f"Invalid epoch count for {name}: {value}") from exc
75	        if epochs <= 0:
76	            raise ValueError(f"Epoch count must be positive for {name}, got {epochs}")
77	        result[name] = epochs
78	    return result

81	def _parse_int_list(spec):
82	    if not spec:
83	        return []
84	    result = []
85	    for item in str(spec).replace(" ", "").split(","):
86	        if not item:
87	            continue
88	        value = int(item)
89	        if value <= 0:
90	            raise ValueError(f"Expected positive integer in list, got {value}")
91	        result.append(value)
92	    return result

95	def _resolve_task_train_iterations(args, task_datasets, steps_per_epoch):
96	    if args.train_budget_mode == "uniform":
97	        return int(args.iterations), None
98	    if args.train_budget_mode != "lada_epochs":
99	        raise ValueError(f"Unsupported train_budget_mode: {args.train_budget_mode}")
100	
101	    epoch_map = dict(LADA_16SHOT_EPOCHS)
102	    epoch_map.update(_parse_epoch_overrides(args.dataset_epoch_overrides))
103	    missing = [d for d in task_datasets if d not in epoch_map]
104	    if missing:
105	        raise ValueError(
106	            "Missing LADA epoch schedule for datasets: "
107	            + ", ".join(missing)
108	            + ". Use --dataset_epoch_overrides dataset=epochs,..."
109	        )
110	    epochs = max(epoch_map[d] for d in task_datasets)
111	    return int(epochs * steps_per_epoch), epochs

120	def _build_text_classifier(args, model, processor, frozen_model, global_class_names,
121	                           cached_seen_text_features, device):
122	    if args.text_classifier_mode == "current":
123	        return get_zeroshot_classifier(model, processor, global_class_names, device)
124	    if args.text_classifier_mode != "lada_hybrid":
125	        raise ValueError(f"Unsupported text_classifier_mode: {args.text_classifier_mode}")
126	
127	    frozen_classifier = get_zeroshot_classifier(frozen_model, processor, global_class_names, device)
128	    if cached_seen_text_features is None or cached_seen_text_features.numel() == 0:
129	        return frozen_classifier
130	    seen_count = cached_seen_text_features.shape[0]
131	    hybrid = frozen_classifier.clone()
132	    hybrid[:, :seen_count] = cached_seen_text_features.to(device).t()
133	    hybrid = hybrid / hybrid.norm(dim=0, keepdim=True)
134	    return hybrid

137	def _fit_spherical_gmm_memory(features, labels, k):
138	    """Fit compact per-class spherical GMM memory from currently accessible features."""
139	    from sklearn.mixture import GaussianMixture
140	
141	    features_np = features.detach().cpu().float().numpy()
142	    labels_cpu = labels.detach().cpu().long()
143	    memory = {}
144	    for cid in sorted(labels_cpu.unique().tolist()):
145	        class_features = features_np[labels_cpu.numpy() == cid]
146	        actual_k = max(1, min(int(k), len(class_features)))
147	        gmm = GaussianMixture(
148	            n_components=actual_k,
149	            covariance_type="spherical",
150	            random_state=42 + int(cid),
151	            reg_covar=1e-6,
152	        )
153	        gmm.fit(class_features)
154	        memory[int(cid)] = {
155	            "means": torch.from_numpy(gmm.means_).float(),
156	            "covariances": torch.from_numpy(gmm.covariances_).float(),
157	            "weights": torch.from_numpy(gmm.weights_).float(),
158	        }
159	    return memory

162	def _sample_spherical_gmm_memory(gmm_memory, samples_per_class, mode, device):
163	    """Sample or repeat pseudo-features from compact spherical GMM memory."""
164	    if not gmm_memory:
165	        raise ValueError("GMM replay requested but no GMM memory is available")
166	    pseudo_features = []
167	    pseudo_labels = []
168	    for cid in sorted(gmm_memory):
169	        entry = gmm_memory[cid]
170	        means = entry["means"].to(device)
171	        covariances = entry["covariances"].to(device)
172	        weights = entry["weights"].to(device)
173	        n_components = means.shape[0]
174	
175	        total = int(samples_per_class)
176	        counts = torch.clamp((weights * total).round().long(), min=1)
177	        diff = total - int(counts.sum().item())
178	        if diff != 0:
179	            counts[int(weights.argmax().item())] += diff
180	        counts = torch.clamp(counts, min=1)
181	
182	        for comp_idx in range(n_components):
183	            count = int(counts[comp_idx].item())
184	            mean = means[comp_idx]
185	            if mode == "gmm_sample":
186	                std = torch.sqrt(torch.clamp(covariances[comp_idx], min=1e-8))
187	                samples = mean.unsqueeze(0) + torch.randn(
188	                    count, mean.shape[0], device=device
189	                ) * std
190	            elif mode == "gmm_mean":
191	                samples = mean.unsqueeze(0).repeat(count, 1)
192	            else:
193	                raise ValueError(f"Unsupported GMM replay mode: {mode}")
194	            samples = F.normalize(samples, dim=-1)
195	            pseudo_features.append(samples)
196	            pseudo_labels.append(
197	                torch.full((count,), int(cid), dtype=torch.long, device=device)
198	            )
199	
200	    return torch.cat(pseudo_features, dim=0), torch.cat(pseudo_labels, dim=0)

203	def _build_center_replay_features(stats_dict, center_means, samples_per_class, device):
204	    """Build compact center replay pseudo-features for classifier-only fine-tuning."""
205	    pseudo_features = []
206	    pseudo_labels = []
207	    for cid in sorted(stats_dict):
208	        if center_means is not None and cid in center_means:
209	            centers = center_means[cid].to(device)
210	        else:
211	            centers = stats_dict[cid].mean.unsqueeze(0).to(device)
212	
213	        n_centers = centers.shape[0]
214	        per_center = max(1, int(samples_per_class) // n_centers)
215	        remainder = int(samples_per_class) - per_center * n_centers
216	        for center_idx in range(n_centers):
217	            count = per_center + (1 if center_idx < remainder else 0)
218	            samples = centers[center_idx].unsqueeze(0).repeat(count, 1)
219	            samples = F.normalize(samples, dim=-1)
220	            pseudo_features.append(samples)
221	            pseudo_labels.append(
222	                torch.full((count,), int(cid), dtype=torch.long, device=device)
223	            )
224	
225	    return torch.cat(pseudo_features, dim=0), torch.cat(pseudo_labels, dim=0)

228	def _compute_dataset_balanced_global_cov(stats_dict, history_class_names):
229	    offset = 0
230	    per_dataset_covs = []
231	    for hcn in history_class_names:
232	        n_classes = len(hcn)
233	        ds_cov = sum(stats_dict[cid].cov for cid in range(offset, offset + n_classes)) / n_classes
234	        per_dataset_covs.append(ds_cov)
235	        offset += n_classes
236	    return sum(per_dataset_covs) / len(per_dataset_covs)

239	def _print_lada_metrics(name, tracker, task_names):
240	    """Print LADA-style Transfer/Average/Last metrics from a full KxK matrix."""
241	    print("\n" + "-" * 110)
242	    print(f"[{name} LADA Metrics]")
243	    header_str = " | ".join([f"{h[:8]:<8}" for h in task_names])
244	    print(f"Metric     | {header_str} | [Average]")
245	    print("-" * 110)
246	
247	    per_task = tracker.calculate_per_task_metrics()
248	    transfer = [per_task[t]["transfer"] for t in task_names]
249	    average = [per_task[t]["average"] for t in task_names]
250	    last = [per_task[t]["last"] for t in task_names]
251	    summary = tracker.get_summary()
252	
253	    print(
254	        "Transfer  | "
255	        + " | ".join(f"{x:8.1f}" if i > 0 else f"{'N/A':>8}" for i, x in enumerate(transfer))
256	        + f" | [{summary['transfer']:.1f}]"
257	    )
258	    print(
259	        "Average   | "
260	        + " | ".join(f"{x:8.1f}" for x in average)
261	        + f" | [{summary['average']:.1f}]"
262	    )
263	    print(
264	        "Last      | "
265	        + " | ".join(f"{x:8.1f}" for x in last)
266	        + f" | [{summary['last']:.1f}]"
267	    )
268	    print("-" * 110)

282	def _atomic_write_json(path, payload):
283	    path = Path(path)
284	    path.parent.mkdir(parents=True, exist_ok=True)
285	    tmp_path = path.with_suffix(path.suffix + ".tmp")
286	    with tmp_path.open("w", encoding="utf-8") as handle:
287	        json.dump(payload, handle, indent=2)
288	    os.replace(tmp_path, path)

291	def _atomic_save_step_artifact(
292	    args,
293	    step_index,
294	    task_name,
295	    model,
296	    cached_seen_text_features,
297	    lr_rgda_classifier,
298	    global_stats_dict,
299	    global_center_means,
300	    dataset_balanced_global_cov,
301	    task_names,
302	    global_class_names,
303	    dataset_label_offsets,
304	    dataset_class_counts,
305	    current_num_classes,
306	    gmm_memory=None,
307	    rgda_stats_by_m=None,
308	):
309	    """Save an immutable post-task snapshot and publish an eval-ready job."""
310	    if not args.async_eval_dir:
311	        raise ValueError("--async_eval_dir is required when saving step artifacts")
312	
313	    base_dir = Path(args.async_eval_dir)
314	    artifact_root = base_dir / "artifacts"
315	    queue_dir = base_dir / "queue"
316	    artifact_root.mkdir(parents=True, exist_ok=True)
317	    queue_dir.mkdir(parents=True, exist_ok=True)
318	
319	    step_name = f"step_{step_index + 1:02d}_{task_name}"
320	    tmp_dir = artifact_root / f".{step_name}.tmp"
321	    final_dir = artifact_root / step_name
322	    if tmp_dir.exists():
323	        shutil.rmtree(tmp_dir)
324	    if final_dir.exists():
325	        shutil.rmtree(final_dir)
326	    tmp_dir.mkdir(parents=True)
327	
328	    artifact_path = tmp_dir / "artifact.pt"
329	    artifact = {
330	        "schema_version": 1,
331	        "created_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
332	        "step_index": step_index,
333	        "task_name": task_name,
334	        "task_names": task_names,
335	        "global_class_names": global_class_names,
336	        "dataset_label_offsets": dataset_label_offsets,
337	        "dataset_class_counts": dataset_class_counts,
338	        "current_num_classes": current_num_classes,
339	        "gmm_memory": gmm_memory,
340	        "args": _serializable_args(args),
341	        "model_state_dict": _cpu_state_dict(model.state_dict()),
342	        "cached_seen_text_features": (
343	            cached_seen_text_features.detach().cpu()
344	            if cached_seen_text_features is not None
345	            else None
346	        ),
347	        "global_stats_dict": global_stats_dict,
348	        "global_center_means": global_center_means,
349	        "dataset_balanced_global_cov": dataset_balanced_global_cov.detach().cpu(),
350	        "rgda_stats_by_m": rgda_stats_by_m,
351	        "lr_rgda": {
352	            "state_dict": _cpu_state_dict(lr_rgda_classifier.classifier.state_dict()),
353	            "num_classes": lr_rgda_classifier.num_classes,
354	            "class_ids": lr_rgda_classifier.class_ids,
355	            "config": {
356	                "rank": args.rgda_rank,
357	                "qda_reg_alpha1": args.rgda_alpha1,
358	                "qda_reg_alpha2": args.rgda_alpha2,
359	                "qda_reg_alpha3": args.rgda_alpha3,
360	                "temperature": 1.0,
361	                "M": args.num_centers,
362	            },
363	        },
364	    }
365	    torch.save(artifact, artifact_path)
366	
367	    os.replace(tmp_dir, final_dir)
368	    ready_payload = {
369	        "schema_version": 1,
370	        "step_index": step_index,
371	        "task_name": task_name,
372	        "artifact_path": str((final_dir / "artifact.pt").resolve()),
373	    }
374	    _atomic_write_json(queue_dir / f"{step_name}.ready.json", ready_payload)
375	    logging.info("Saved async eval artifact: %s", final_dir)

635	def main(args):
636	    if args.seed is not None:
637	        fix_random_seed(args.seed)
638	
639	    # ========== 1. 初始化 ==========
640	    logging.info("\n=== Initializing Incremental Learning ===")
641	    trainer = LoRANSPTrainer(args)
642	    model = trainer.model
643	    processor = trainer.processor
644	
645	    # 解析文本编码器调度的默认值
646	    if args.text_tuning_schedule is None:
647	        args.text_tuning_schedule = "always" if args.tune_text_encoder else "never"
648	
649	    # 验证文本调度参数
650	    if args.text_schedule_switch_task < 1:
651	        raise ValueError("--text_schedule_switch_task must be >= 1")
652	    if args.text_lr_scale_after_task < 0:
653	        raise ValueError("--text_lr_scale_after_task must be >= 0")
654	
655	    # 非 never 调度时强制启用文本 LoRA（否则调度无意义）
656	    if args.text_tuning_schedule == "never":
657	        args.tune_text_encoder = False
658	    else:
659	        args.tune_text_encoder = True
660	
661	    use_distillation = args.fd_weight > 0 or args.cd_weight > 0
662	    reference_loader = load_reference_dataset(args, trainer.model_pretrain,
663	                                              processor, args.device) if use_distillation else None
664	
665	    # 预收集所有数据集的类名，用于全局 ZS 分类器（text LoRA 每次 merge 后更新）
666	    global_class_names = []
667	    task_names = [d[0] for d in args.dataset_sequence]
668	    dataset_label_offsets = {}
669	    dataset_class_counts = {}
670	    next_label_offset = 0
671	    for task_datasets in args.dataset_sequence:
672	        for d_name in task_datasets:
673	            c_names = get_xtail_classnames(args.root, d_name, args.num_shots)
674	            dataset_label_offsets[d_name] = next_label_offset
675	            dataset_class_counts[d_name] = len(c_names)
676	            next_label_offset += len(c_names)
677	            global_class_names.extend(c_names)
678	
679	    # ========== 2. 增量学习循环 ==========
680	    history_class_names = []  # 记录所有已学类名列表的列表
681	    cached_seen_text_features = None
682	    artifact_num_centers = sorted(set([args.num_centers] + _parse_int_list(args.artifact_num_centers)))
683	    global_rgda_stats_by_m = {m: {} for m in artifact_num_centers}
684	    global_rgda_centers_by_m = {m: None for m in artifact_num_centers}
685	    dataset_balanced_cov_by_m = {}
686	    global_stats_dict = {}
687	    global_center_means = None  # 多中心 LR-RGDA 的类中心映射，用局部变量替代 main._global_center_means
688	    global_gmm_memory = {}
689	
690	    metrics_zs = ContinualLearningMetrics(task_names)
691	    metrics_rgda = ContinualLearningMetrics(task_names)
692	    metrics_ens = ContinualLearningMetrics(task_names)
693	    acc_matrix_lada = [] if args.enable_lada else None
694	    acc_matrix_lada_zs = [] if args.enable_lada else None
695	
696	    for i, task_datasets in enumerate(args.dataset_sequence):
697	        print(f"\n" + "=" * 50)
698	        print(f"=== Task {i+1}: {task_datasets} ===")
699	        print("=" * 50)
700	
701	        # --- 2a. 准备训练数据 ---
702	        train_loaders = []
703	        task_class_names = []
704	        for d_name in task_datasets:
705	            train_transform, test_transform = get_transforms(d_name)
706	            num_shots = None if args.full_shot else args.num_shots
707	            tr_loader, _, _, c_names = get_xtail_trainloader(
708	                root=args.root, dataset_name=d_name,
709	                transform_train=train_transform, transform_test=test_transform,
710	                num_shots=num_shots, batch_size=args.batch_size
711	            )
712	            train_loaders.append(tr_loader)
713	            task_class_names.extend(c_names)
714	
715	        merged_dataset = ConcatDataset([loader.dataset for loader in train_loaders])
716	        # 用于提取协方差的 loader 不打乱顺序
717	        cov_loader = DataLoader(merged_dataset, batch_size=args.batch_size, shuffle=False)
718	        merged_loader = DataLoader(merged_dataset, batch_size=args.batch_size, shuffle=True)
719	        task_train_iterations, task_epochs = _resolve_task_train_iterations(
720	            args, task_datasets, len(merged_loader))
721	        if task_epochs is None:
722	            logging.info(
723	                "Task %d training budget: %d iterations (uniform)",
724	                i + 1, task_train_iterations,
725	            )
726	        else:
727	            logging.info(
728	                "Task %d training budget: %d epochs x %d steps/epoch = %d iterations",
729	                i + 1, task_epochs, len(merged_loader), task_train_iterations,
730	            )
731	
732	        # --- 2b. 初始化适配器（非标准 LoRA 时在训练前初始化） ---
733	        if args.init_mode not in ["lora_nsp", "lora_vanilla"]:
734	            logging.info(f"\n=== Initializing adapters ({args.init_mode}) ===")
735	            if "proj_sigma" in args.init_mode and args.init_mode != "lora_nsp":
736	                window = "tail" if "tail" in args.init_mode else "middle"
737	                if trainer.covariance_history:
738	                    trainer.model.vision_model.initialize_adapters_from_covariance(
739	                        trainer.covariance_history, window=window)
740	                else:
741	                    logging.info("No covariance history yet, using default random init for Task 1")
742	            elif "weight_svd" in args.init_mode:
743	                window = "tail" if "tail" in args.init_mode else "middle"
744	                trainer.model.vision_model.initialize_adapters_from_weight_svd(window=window)
745	
746	        # --- 2c. 训练模型 ---
747	        # 文本编码器调度：根据当前任务决定 text_lr
748	        task_number = i + 1
749	        schedule = args.text_tuning_schedule
750	        text_lr = args.lr  # 默认
751	        if schedule == "never":
752	            text_lr = 0.0
753	        elif schedule in ("freeze_after", "low_lr_after"):
754	            if task_number > args.text_schedule_switch_task:
755	                if schedule == "freeze_after":
756	                    text_lr = 0.0
757	                else:
758	                    text_lr = args.lr * args.text_lr_scale_after_task
759	        train_text_this_task = text_lr > 0
760	        model = trainer.train(merged_loader, task_class_names, reference_loader,
761	                              aux_weight=args.aux_weight,
762	                              train_text_encoder=train_text_this_task,
763	                              text_lr=text_lr,
764	                              iterations=task_train_iterations)
765	
766	        # --- 2d. 任务后处理：合入 + 协方差累积 ---
767	        extracted_covariances = None
768	        if args.init_mode == "lora_nsp":
769	            print("\n=== Applying Null-Space Projection (NSP) ===")
770	            # 图像编码器 NSP
771	            text_covariances = None
772	            if trainer.has_vision_lora:
773	                extracted_covariances = trainer.extract_layer_covariances(cov_loader)
774	            # 文本编码器 NSP（仅当本轮实际训练了文本编码器）
775	            if trainer.has_text_lora and train_text_this_task:
776	                text_covariances = trainer.extract_text_covariances(task_class_names)
777	            trainer.finalize_task_for_incremental()
778	            if trainer.has_vision_lora:
779	                trainer.update_covariance_history(extracted_covariances)
780	            if trainer.has_text_lora and train_text_this_task:
781	                trainer.update_text_covariance_history(text_covariances)
782	        elif "proj_sigma" in args.init_mode:
783	            print("\n=== Proj-Σ: Extracting covariances + merging ===")
784	            if trainer.has_vision_lora:
785	                extracted_covariances = trainer.extract_layer_covariances(cov_loader)
786	                trainer.update_covariance_history(extracted_covariances, update_projection=False)
787	            trainer.finalize_task_for_incremental()
788	        else:
789	            print(f"\n=== Merging LoRA Weights (init_mode={args.init_mode}) ===")
790	            trainer.finalize_task_for_incremental()
791	
792	        # --- 2d-bis. New basis/null-init modes (after cov extraction + merge) ---
793	        if trainer.has_vision_lora and args.projection_param_mode in ("fixed_basis", "core_basis"):
794	            cov_source = extracted_covariances if extracted_covariances else trainer.covariance_history
795	            if cov_source:
796	                if args.init_mode != "lora_nsp":
797	                    # lora_nsp path already set basis_U via update_projection_matrices
798	                    trainer.model.vision_model.set_basis_from_covariance(
799	                        cov_source, args.basis_rank, window=args.basis_window)
800	
801	        if trainer.has_vision_lora and args.null_init_mode != "none":
802	            cov_source = extracted_covariances if extracted_covariances else trainer.covariance_history
803	            if cov_source:
804	                logging.info(f"\n=== LoRA-Null Init (mode={args.null_init_mode}) ===")
805	                trainer.model.vision_model.initialize_history_null(
806	                    cov_source, window=args.basis_window)
807	
808	        if args.text_classifier_mode == "lada_hybrid":
809	            task_text_features = _encode_text_classifier_columns(
810	                model, processor, task_class_names, args.device)
811	            cached_seen_text_features = (
812	                task_text_features.detach()
813	                if cached_seen_text_features is None
814	                else torch.cat([cached_seen_text_features, task_text_features.detach()], dim=0)
815	            )
816	            logging.info(
817	                "Cached LADA-style seen text prototypes: %s",
818	                tuple(cached_seen_text_features.shape),
819	            )
820	            if args.text_adapter_type == "lada_adaptformer" and hasattr(model.text_model, "reset_adapters"):
821	                model.text_model.reset_adapters()
822	                logging.info("Reset LADA text AdaptFormer after caching current task prototypes.")
823	
824	        # --- 2e. 提取特征并构建统计字典 ---
825	        task_features = []
826	        task_labels = []
827	        task_gmm_features = []
828	        task_gmm_labels = []
829	        label_offset = sum(len(c_names) for c_names in history_class_names)
830	
831	        for d_name in task_datasets:
832	            train_transform, test_transform = get_transforms(d_name)
833	            tr_loader, tr4update, _, c_names = get_xtail_trainloader(
834	                root=args.root, dataset_name=d_name,
835	                transform_train=train_transform, transform_test=test_transform,
836	                num_shots=num_shots, batch_size=args.batch_size
837	            )
838	            # 根据 --classifier_feature_transform 选择特征提取的 transform
839	            # 'test' = 确定性变换（publication protocol），'train' = 随机增强
840	            feature_loader = tr_loader if args.classifier_feature_transform == "train" else tr4update
841	            from src.utils.feature_extractor import extract_features
842	            raw_features, labels = extract_features(
843	                model, feature_loader, args.device, normalize=False)
844	            features = F.normalize(raw_features, dim=-1)
845	            global_labels = labels + label_offset
846	            task_features.append(features)
847	            task_labels.append(global_labels)
848	            gmm_features = raw_features if args.gmm_fit_space == "raw" else features
849	            task_gmm_features.append(gmm_features)
850	            task_gmm_labels.append(global_labels)
851	
852	        task_features = torch.cat(task_features)
853	        task_labels = torch.cat(task_labels)
854	        task_gmm_features = torch.cat(task_gmm_features)
855	        task_gmm_labels = torch.cat(task_gmm_labels)
856	
857	        # 构建一套或多套 compact LR-RGDA stats。主 M 用于 inline classifier；
858	        # 其它 M 只写入 artifact，供离线 classifier sweep 使用。
859	        for m in artifact_num_centers:
860	            task_stats_dict_m, center_means_m = build_multi_center_stats_dict(
861	                task_features, task_labels, M=m)
862	            global_rgda_stats_by_m[m].update(task_stats_dict_m)
863	            if center_means_m is not None:
864	                if global_rgda_centers_by_m[m] is None:
865	                    global_rgda_centers_by_m[m] = {}
866	                # center_means 的 key 已是全局类ID（来自 task_labels + label_offset）
867	                global_rgda_centers_by_m[m].update(center_means_m)
868	
869	        global_stats_dict = global_rgda_stats_by_m[args.num_centers]
870	        global_center_means = global_rgda_centers_by_m[args.num_centers]
871	
872	        task_gmm_memory = _fit_spherical_gmm_memory(
873	            task_gmm_features, task_gmm_labels, k=args.gmm_k)
874	        global_gmm_memory.update(task_gmm_memory)
875	        logging.info(
876	            "[GMM Memory] Updated spherical GMM memory: %d seen classes, "
877	            "k=%d, fit_space=%s",
878	            len(global_gmm_memory), args.gmm_k, args.gmm_fit_space,
879	        )
880	
881	        history_class_names.append(task_class_names)
882	
883	        # --- 2f. 构建分类器 ---
884	        # 计算数据集等权的全局协方差
885	        dataset_balanced_cov_by_m = {
886	            m: _compute_dataset_balanced_global_cov(stats_dict_m, history_class_names)
887	            for m, stats_dict_m in global_rgda_stats_by_m.items()
888	        }
889	        dataset_balanced_global_cov = dataset_balanced_cov_by_m[args.num_centers]
890	
891	        lr_rgda_classifier = LRRGDAClassifier(
892	            stats_dict=global_stats_dict,
893	            device=args.device,
894	            rank=args.rgda_rank,
895	            M=args.num_centers,
896	            center_means=global_center_means,
897	            qda_reg_alpha1=args.rgda_alpha1,
898	            qda_reg_alpha2=args.rgda_alpha2,
899	            qda_reg_alpha3=args.rgda_alpha3,
900	            temperature=1.0,
901	            global_cov=dataset_balanced_global_cov,
902	        )
903	        # 分类器微调（如果启用）：LR-RGDA 仍由正常 mean/cov 构建；
904	        # 这里只使用 compact replay pseudo-features 微调 classifier 参数。
905	        if args.rgda_train_iter > 0:
906	            if args.rgda_fit_source == "none":
907	                logging.info(
908	                    "[Fit] --rgda_train_iter=%d but --rgda_fit_source=none; "
909	                    "skipping classifier fine-tuning.",
910	                    args.rgda_train_iter,
911	                )
912	                replay_features = replay_labels = None
913	            elif args.rgda_fit_source == "center_replay":
914	                replay_features, replay_labels = _build_center_replay_features(
915	                    global_stats_dict,
916	                    global_center_means,
917	                    args.gaussian_samples_per_class,
918	                    args.device,
919	                )
920	            elif args.rgda_fit_source in ("gmm_mean", "gmm_sample"):
921	                replay_features, replay_labels = _sample_spherical_gmm_memory(
922	                    global_gmm_memory,
923	                    args.gaussian_samples_per_class,
924	                    args.rgda_fit_source,
925	                    args.device,
926	                )
927	            else:
928	                raise ValueError(f"Unsupported rgda_fit_source: {args.rgda_fit_source}")
929	
930	            if replay_features is not None:
931	                logging.info(
932	                    "[Fit] Fine-tuning LR-RGDA for %d iters on %d compact replay "
933	                    "pseudo-features (fit_source=%s, samples_per_class=%d)...",
934	                    args.rgda_train_iter,
935	                    replay_features.shape[0],
936	                    args.rgda_fit_source,
937	                    args.gaussian_samples_per_class,
938	                )
939	                lr_rgda_classifier.fit(
940	                    replay_features, replay_labels,
941	                    iterations=args.rgda_train_iter,
942	                    lr=args.rgda_train_lr,
943	                )
944	
945	        current_num_classes = sum(len(c_names) for c_names in history_class_names)
946	        if args.save_step_artifacts:
947	            rgda_stats_by_m = {
948	                int(m): {
949	                    "global_stats_dict": global_rgda_stats_by_m[m],
950	                    "global_center_means": global_rgda_centers_by_m[m],
951	                    "dataset_balanced_global_cov": dataset_balanced_cov_by_m[m].detach().cpu(),
952	                }
953	                for m in artifact_num_centers
954	            }
955	            _atomic_save_step_artifact(
956	                args=args,
957	                step_index=i,
958	                task_name=task_datasets[0],
959	                model=model,
960	                cached_seen_text_features=cached_seen_text_features,
961	                lr_rgda_classifier=lr_rgda_classifier,
962	                global_stats_dict=global_stats_dict,
963	                global_center_means=global_center_means,
964	                dataset_balanced_global_cov=dataset_balanced_global_cov,
965	                task_names=task_names,
966	                global_class_names=global_class_names,
967	                dataset_label_offsets=dataset_label_offsets,
968	                dataset_class_counts=dataset_class_counts,
969	                current_num_classes=current_num_classes,
970	                gmm_memory=global_gmm_memory,
971	                rgda_stats_by_m=rgda_stats_by_m,
972	            )
973	
974	        if args.skip_inline_eval:
975	            continue
976	
977	        zeroshot_classifier = _build_text_classifier(
978	            args, model, processor, trainer.model_pretrain, global_class_names,
979	            cached_seen_text_features, args.device)
980	
981	        # --- 2g. 按 LADA 协议评估所有任务 ---
982	        print("\n=== Evaluating Task ===")
983	        step_accs_zs, step_accs_rgda, step_accs_ens = {}, {}, {}
984	
985	        for j in range(len(args.dataset_sequence)):
986	            eval_datasets = args.dataset_sequence[j]
987	            d_name = eval_datasets[0]  # 每个 Task 只有一个数据集
988	            eval_label_offset = dataset_label_offsets[d_name]
989	
990	            zs_acc, rgda_acc, ens_acc, lada_acc, lada_zs_acc, c_len, _ = evaluate_dataset(
991	                args, d_name, model, zeroshot_classifier, lr_rgda_classifier,
992	                current_num_classes, eval_label_offset
993	            )
994	            expected_c_len = dataset_class_counts[d_name]
995	            if c_len != expected_c_len:
996	                raise ValueError(
997	                    f"Class count mismatch for {d_name}: eval={c_len}, expected={expected_c_len}"
998	                )
999	
1000	            print(f"[Tested on Task {j+1}: {d_name:<10s}] -> "
1001	                  f"Zero-shot: {zs_acc:5.1f}% | LR-RGDA: {rgda_acc:5.1f}% | "
1002	                  f"Ensemble: {ens_acc:5.1f}%")
1003	
1004	            step_accs_zs[d_name] = zs_acc / 100.0
1005	            step_accs_rgda[d_name] = rgda_acc / 100.0
1006	            step_accs_ens[d_name] = ens_acc / 100.0
1007	
1008	        metrics_zs.update(i, step_accs_zs)
1009	        metrics_rgda.update(i, step_accs_rgda)
1010	        metrics_ens.update(i, step_accs_ens)
1011	
1012	    if args.skip_inline_eval:
1013	        logging.info(
1014	            "Skipped inline evaluation. Use scripts/evaluate_incremental_artifacts.py "
1015	            "and scripts/merge_incremental_async_results.py with --async_eval_dir=%s",
1016	            args.async_eval_dir,
1017	        )
